{{- /* Generic experiment embedder. The argument is an experiment name. This reads experiments//manifest.yaml — the experiment's own declaration of its mount element, stylesheets, web fonts, third-party dependencies, and scripts — and assembles the page from it: compile + link styles, fetch each declared font and dependency at build time (verifying the pinned SRI) and serve it self-hosted, then load the local scripts in order. The host hardcodes nothing about any specific experiment. Usage: {{ partial "experiments/embed.html" "llm-explorer" }} */ -}} {{- $name := . -}} {{- $dir := printf "experiments/%s" $name -}} {{- $m := resources.Get (printf "%s/manifest.yaml" $dir) | transform.Unmarshal -}} {{- /* Host baseline: a clean canvas (neutralizes inherited theme styles) shared by every experiment, so individual experiments don't re-fix them. Linked first so an experiment's own styles can still override it. */ -}} {{- $base := resources.Get "styles/experiment-base.scss" | toCSS -}} {{- if hugo.IsServer -}} {{- else -}} {{- $base = $base | minify | fingerprint -}} {{- end -}} {{- /* Stylesheets — compile .scss; fingerprint in production builds. */ -}} {{- range $m.styles -}} {{- $res := resources.Get (printf "%s/%s" $dir .) -}} {{- if strings.HasSuffix . ".scss" }}{{ $res = $res | toCSS }}{{ end -}} {{- if hugo.IsServer -}} {{- else -}} {{- $res = $res | minify | fingerprint -}} {{- end -}} {{- end -}} {{- /* Web fonts — treated like third-party script dependencies: fetched at build time, SRI-verified against the manifest's pin, and served self-hosted via @font-face so both the experiment's CSS and its canvas can use them. */ -}} {{- range $m.fonts -}} {{- $url := .url -}} {{- $want := .integrity -}} {{- $font := "" -}} {{- with try (resources.GetRemote $url) -}} {{- with .Err }}{{ errorf "experiment %q: fetching font %s failed: %s" $name $url . }}{{ else }}{{ $font = .Value }}{{ end -}} {{- end -}} {{- $font = $font | fingerprint "sha256" -}} {{- if $want }}{{ if ne $font.Data.Integrity $want }}{{ errorf "experiment %q: integrity mismatch for %s (manifest %q, built %q)" $name $url $want $font.Data.Integrity }}{{ end }}{{ end -}} {{- end -}} {{- /* Mount element: the app builds itself into this. The shared `experiment-embed` class carries the host baseline (clean canvas + generous width); the `{{ $m.mount }}` class carries this experiment's own styles (convention: mount id == that class). */ -}}
{{- /* Third-party dependencies — fetched at build time, SRI-verified, served self-hosted (no runtime CDN dependency). Loaded before local scripts. */ -}} {{- range $m.dependencies -}} {{- $url := .url -}} {{- $want := .integrity -}} {{- $dep := "" -}} {{- with try (resources.GetRemote $url) -}} {{- with .Err }}{{ errorf "experiment %q: fetching %s failed: %s" $name $url . }}{{ else }}{{ $dep = .Value }}{{ end -}} {{- end -}} {{- $dep = $dep | fingerprint "sha256" -}} {{- if $want }}{{ if ne $dep.Data.Integrity $want }}{{ errorf "experiment %q: integrity mismatch for %s (manifest %q, built %q)" $name $url $want $dep.Data.Integrity }}{{ end }}{{ end -}} {{- end -}} {{- /* Local scripts — in declared order (app entry last). */ -}} {{- range $m.scripts -}} {{- $res := resources.Get (printf "%s/%s" $dir .) -}} {{- if hugo.IsServer -}} {{- else -}} {{- $res = $res | minify | fingerprint -}} {{- end -}} {{- end -}}
